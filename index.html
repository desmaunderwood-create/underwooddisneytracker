    // --- DATA MANAGEMENT ---

    function loadSavedData() {

      try {

        const saved = localStorage.getItem(STORAGE_KEY);

        if (saved) {

          appState.userProgress = JSON.parse(saved);

        }

      } catch (e) {

        console.error('Failed to load local storage data:', e);

      }

    }


    function saveData() {

      try {

        localStorage.setItem(STORAGE_KEY, JSON.stringify(appState.userProgress));

        updateDashboardStats();

      } catch (e) {

        console.error('Failed to save data:', e);

      }

    }


    // Toggle Checkbox

    function toggleMember(attractionId, member) {

      if (!appState.userProgress[attractionId]) {

        appState.userProgress[attractionId] = { greg: false, dez: false, emma: false, harper: false };

      }

      appState.userProgress[attractionId][member] = !appState.userProgress[attractionId][member];

      

      saveData();

      updateAttractionCardState(attractionId);

    }


    // Quick Check All / Uncheck All for an Attraction

    function toggleAllForAttraction(attractionId) {

      const current = appState.userProgress[attractionId] || {};

      const allChecked = FAMILY_MEMBERS.every(m => current[m]);


      appState.userProgress[attractionId] = {

        greg: !allChecked,

        dez: !allChecked,

        emma: !allChecked,

        harper: !allChecked

      };


      saveData();

      renderApp();

    }


    // --- DASHBOARD & STATS CALCULATIONS ---

    function updateDashboardStats() {

      let totalAttractions = 0;

      let totalCompletedAtLeastOne = 0;

      let allFourCompleted = 0;


      const memberCounts = { greg: 0, dez: 0, emma: 0, harper: 0 };


      PARKS_DATA.forEach(park => {

        park.lands.forEach(land => {

          land.attractions.forEach(attr => {

            totalAttractions++;

            const status = appState.userProgress[attr.id] || {};

            

            let riddenByAny = false;

            let riddenByAll = true;


            FAMILY_MEMBERS.forEach(member => {

              if (status[member]) {

                memberCounts[member]++;

                riddenByAny = true;

              } else {

                riddenByAll = false;

              }

            });


            if (riddenByAny) totalCompletedAtLeastOne++;

            if (riddenByAll) allFourCompleted++;

          });

        });

      });


      // Update DOM

      document.getElementById('total-attractions-count').textContent = totalAttractions;

      document.getElementById('total-completed-count').textContent = totalCompletedAtLeastOne;

      document.getElementById('all-four-count').textContent = `${allFourCompleted} / ${totalAttractions}`;


      const pctOverall = totalAttractions > 0 ? (totalCompletedAtLeastOne / totalAttractions) * 100 : 0;

      document.getElementById('overall-progress-bar').style.width = `${pctOverall}%`;


      // Update Individual Member Stats

      FAMILY_MEMBERS.forEach(m => {

        const count = memberCounts[m];

        document.getElementById(`stat-${m}-count`).textContent = `${count}/${totalAttractions}`;

        const pct = totalAttractions > 0 ? (count / totalAttractions) * 100 : 0;

        document.getElementById(`bar-${m}`).style.width = `${pct}%`;

      });

    }

    // --- RENDER LOGIC ---

    function renderApp() {

      updateDashboardStats();


      const container = document.getElementById('attractions-container');

      container.innerHTML = '';


      let parksToDisplay = PARKS_DATA;

      if (appState.activePark !== 'all') {

        parksToDisplay = PARKS_DATA.filter(p => p.id === appState.activePark);

      }


      let totalRenderedAttractions = 0;


      parksToDisplay.forEach(park => {

        const parkWrapper = document.createElement('div');

        parkWrapper.className = 'space-y-4';


        let parkHeaderHTML = `

          <div class="flex items-center justify-between border-b border-sage-light/80 pb-2 pt-2">

            <h2 class="text-base sm:text-lg font-bold text-forest flex items-center space-x-2">

              <i class="fa-solid ${park.icon} ${park.color}"></i>

              <span>${park.name}</span>

            </h2>

          </div>

        `;


        let parkAttractionsCount = 0;

        let landsHTML = '';


        park.lands.forEach(land => {

          const filteredAttractions = land.attractions.filter(attr => {

            const status = appState.userProgress[attr.id] || {};

            

            // Search Query Filter

            if (appState.searchQuery) {

              const q = appState.searchQuery;

              const matchesName = attr.name.toLowerCase().includes(q);

              const matchesLand = land.name.toLowerCase().includes(q);

              if (!matchesName && !matchesLand) return false;

            }


            // Category Filters

            if (appState.activeFilter === 'unridden') {

              return !FAMILY_MEMBERS.some(m => status[m]);

            }

            if (appState.activeFilter === 'all4') {

              return FAMILY_MEMBERS.every(m => status[m]);

            }

            if (appState.activeFilter.startsWith('not-')) {

              const targetMember = appState.activeFilter.replace('not-', '');

              return !status[targetMember];

            }


            return true;

          });


          if (filteredAttractions.length > 0) {

            parkAttractionsCount += filteredAttractions.length;

            totalRenderedAttractions += filteredAttractions.length;


            landsHTML += `

              <div class="space-y-2">

                <div class="text-xs font-bold text-sage-dark uppercase tracking-wider pl-1">${land.name}</div>

                <div class="grid grid-cols-1 gap-2.5">

                  ${filteredAttractions.map(attr => createAttractionCardHTML(attr)).join('')}

                </div>

              </div>

            `;

          }

        });


        if (parkAttractionsCount > 0) {

          parkWrapper.innerHTML = parkHeaderHTML + landsHTML;

          container.appendChild(parkWrapper);

        }

      });


      // Update Result Count Label

      document.getElementById('results-count').textContent = `Showing ${totalRenderedAttractions} attractions`;


      // Active Park Header Title Label

      const titleElem = document.getElementById('park-summary-title');

      if (appState.activePark === 'all') {

        titleElem.textContent = 'All Parks & Experiences';

      } else {

        const found = PARKS_DATA.find(p => p.id === appState.activePark);

        titleElem.textContent = found ? found.name : 'Attraction List';

      }


      // Empty State

      if (totalRenderedAttractions === 0) {

        container.innerHTML = `

          <div class="text-center py-12 px-4 bg-white rounded-2xl border border-sage-light">

            <div class="w-12 h-12 rounded-full bg-sage-light/50 text-sage-dark mx-auto flex items-center justify-center mb-3 text-lg">

              <i class="fa-solid fa-compass"></i>

            </div>

            <h3 class="font-bold text-slate-700 text-sm">No Attractions Found</h3>

            <p class="text-xs text-slate-500 mt-1">Try adjusting your filters or search term to see more results.</p>

          </div>

        `;

      }

    }

      openConfirmModal(

        'Reset Entire App Data?',

        'This will erase ALL tracked attractions for all family members.',

        () => {

          appState.userProgress = {};

          saveData();

          renderApp();

          showToast('All progress reset');

        }

      );

    }


    function openConfirmModal(title, msg, onConfirm) {

      document.getElementById('confirm-title').textContent = title;

      document.getElementById('confirm-msg').textContent = msg;

      confirmCallback = onConfirm;

      document.getElementById('confirm-modal').classList.remove('hidden');

    }


    function closeConfirmModal(result) {

      document.getElementById('confirm-modal').classList.add('hidden');

      if (result && confirmCallback) {

        confirmCallback();

      }

      confirmCallback = null;

    }


    // --- EXPORT & IMPORT ---

    function exportData() {

      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(appState.userProgress, null, 2));

      const downloadAnchor = document.createElement('a');

      downloadAnchor.setAttribute("href", dataStr);

      downloadAnchor.setAttribute("download", `wdw_family_tracker_backup_${new Date().toISOString().slice(0,10)}.json`);

      document.body.appendChild(downloadAnchor);

      downloadAnchor.click();

      downloadAnchor.remove();

      showToast('Backup downloaded');

    }


    function importData(event) {

      const file = event.target.files[0];

      if (!file) return;


      const reader = new FileReader();

      reader.onload = function(e) {

        try {

          const parsed = JSON.parse(e.target.result);

          if (typeof parsed === 'object' && parsed !== null) {

            appState.userProgress = parsed;

            saveData();

            renderApp();

            closeSettingsModal();

            showToast('Backup restored successfully!');

          } else {

            throw new Error('Invalid JSON format');

          }

        } catch (err) {

          alert('Failed to import backup file. Please ensure it is a valid backup JSON.');

        }

      };

      reader.readAsText(file);

    }


    function showToast(message) {

      const toast = document.getElementById('toast');

      document.getElementById('toast-message').textContent = message;

      toast.classList.remove('opacity-0', 'pointer-events-none', 'translate-y-4');

      

      setTimeout(() => {

        toast.classList.add('opacity-0', 'pointer-events-none', 'translate-y-4');

      }, 2500);

    }

    function setupPWA() {

      // Manifest Blob creation

      const manifest = {

        name: "WDW Family Tracker",

        short_name: "WDW Tracker",

        start_url: ".",

        display: "standalone",

        background_color: "#FAF9F5",

        theme_color: "#1B3B2B",

        icons: [

          {

            src: "https://placehold.co/192x192/1B3B2B/D4AF37?text=WDW",

            sizes: "192x192",

            type: "image/png"

          },

          {

            src: "https://placehold.co/512x512/1B3B2B/D4AF37?text=WDW",

            sizes: "512x512",

            type: "image/png"

          }

        ]

      };

      

      const stringManifest = JSON.stringify(manifest);

      const blob = new Blob([stringManifest], {type: 'application/json'});

      const manifestURL = URL.createObjectURL(blob);

      const link = document.createElement('link');

      link.rel = 'manifest';

      link.href = manifestURL;

      document.head.appendChild(link);


      // Register offline service worker blob

      if ('serviceWorker' in navigator) {

        const swCode = `

          const CACHE_NAME = 'wdw-tracker-cache-v1';

          self.addEventListener('install', (e) => {

            e.waitUntil(

              caches.open(CACHE_NAME).then((cache) => {

                return cache.addAll(['./']);

              })

            );

          });

          self.addEventListener('fetch', (e) => {

            e.respondWith(

              caches.match(e.request).then((res) => res || fetch(e.request))

            );

          });

        `;

        const swBlob = new Blob([swCode], { type: 'text/javascript' });

        const swUrl = URL.createObjectURL(swBlob);

        navigator.serviceWorker.register(swUrl).catch(err => console.log('SW registration skipped:', err));

      }

    }
